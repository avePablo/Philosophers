void take_forks(t_philosophers *p)
{
    if (p->id % 2 == 0)
    {
        pthread_mutex_lock(&p->data->forks[p->fork_r]);
        print_state(p, "has taken the right fork");
        pthread_mutex_lock(&p->data->forks[p->fork_l]);
        print_state(p, "has taken the left fork");
    }
    else
    {
        pthread_mutex_lock(&p->data->forks[p->fork_l]);
        print_state(p, "has taken the left fork");
        pthread_mutex_lock(&p->data->forks[p->fork_r]);
        print_state(p, "has taken the right fork");
    }
}


void release_forks(t_philosophers *p)
{
    pthread_mutex_unlock(&p->data->forks[p->fork_l]);
    pthread_mutex_unlock(&p->data->forks[p->fork_r]);
}

void eat(t_philosophers *p)
{   
    if (all_ate(p->data) == 0)
    {
    print_state(p, "is eating");
    p->last_meal = timestamp_ms();
    p->times_eaten++;
    smart_sleep(p->data->time_to_eat, p->data);
    }
}

void sleep_philo(t_philosophers *p)
{
    print_state(p, "is sleeping");
   smart_sleep(p->data->time_to_sleep, p->data);

}

void think(t_philosophers *p)
{
    print_state(p, "is thinking");
}

int is_digit_str(char *str)
{
    int i = 0;
    if (!str || !str[0])
        return 1;
    while (str[i])
    {
        if (str[i] < '0' || str[i] > '9')
            return 1;
        i++;
    }
    return 0;
}

static int	is_whitespace(char c)
{
	return (c == ' ' || c == '\t' || c == '\n' || c == '\v' || c == '\f'
		|| c == '\r');
}

static long	handle_overflow(long result, int sign, char next_digit)
{
	if (result > LONG_MAX / 10 || (result == LONG_MAX / 10 && next_digit > '7'))
	{
		if (sign == 1)
			return (LONG_MAX);
		else
			return (LONG_MIN);
	}
	return (0);
}

long	ft_atol(char *str)
{
	int		i;
	long	result;
	long	overflow_check;
	int		sign;

	i = 0;
	sign = 1;
	result = 0;
	while (is_whitespace(str[i]))
		i++;
	if (str[i] == '-' || str[i] == '+')
	{
		if (str[i] == '-')
			sign = -1;
		i++;
	}
	while (str[i] >= '0' && str[i] <= '9')
	{
		overflow_check = handle_overflow(result, sign, str[i]);
		if (overflow_check)
			return (overflow_check);
		result = result * 10 + (str[i++] - '0');
	}
	return (result * sign);
}

void init_philos(t_philosophers *philo, t_general *data)
{
    int i = 0;
    while (i < data->number_of_philosophers)
    {
        philo[i].id = i + 1;
        philo[i].fork_l = i;
        philo[i].fork_r = (i + 1) % data->number_of_philosophers;
        philo[i].last_meal = timestamp_ms();
        philo[i].data = data;
        philo[i].times_eaten = 0;
        i++;
    }
}

void init_general(t_general *data, char *argv[])
{
    data->number_of_philosophers = ft_atol(argv[1]);
    data->time_to_die = ft_atol(argv[2]);
    data->time_to_eat = ft_atol(argv[3]);
    data->time_to_sleep = ft_atol(argv[4]);
    data->dead = 0;
    data->number_of_times_each_philosopher_must_eat = (argv[5]) ? atol(argv[5]) : -1;
    data->start_time = timestamp_ms();

    pthread_mutex_init(&data->print, NULL);

    int i = 0;
    while (i < data->number_of_philosophers)
    {
        pthread_mutex_init(&data->forks[i], NULL);
        i++;
    }
}

void create_philo_threads(t_general *data)
{
    int i = 0;
    while (i < data->number_of_philosophers)
    {
        pthread_create(&data->philo[i].thread, NULL, ft_routine, &data->philo[i]);
        i++;
    }
}

void join_philo_threads(t_general *data, pthread_t monitor_thread)
{
    int i = 0;
    while (i < data->number_of_philosophers)
    {
        pthread_join(data->philo[i].thread, NULL);
        i++;
    }
    pthread_join(monitor_thread, NULL);
}

int main(int argc, char **argv)
{
    int i = 0;
    pthread_t monitor_thread;
    
    i = parse(argc, argv);
    if(i != 0 && i < 3)
    {
        error_handler(i);
        return 1;
    }
    t_general *data = malloc(sizeof(t_general));
    data->philo = malloc(sizeof(t_philosophers) * ft_atol(argv[1]));
    init_general(data, argv);
    init_philos(data->philo, data);
    if (i == 3)
    {
        one_philo_only(data);
        return 1;
    }
    print_philos(data);
    create_philo_threads(data);
    pthread_create(&monitor_thread, NULL, monitor, data);
    join_philo_threads(data, monitor_thread);
    free(data->philo);
    free(data);
    return 0;
}#include "philosophers.h"

int parse(int argc, char** argv)
{   
    long arg;
    int i;
    i = 1;
    if (argc < 5 || argc > 6)
        return 1;
    while(i < argc)
    {
        arg = ft_atol(argv[i]);
        if (is_digit_str(argv[i]) == 1)
        return 1;
        if (i == 1 && (arg == 1))
        {
        return 3;
        }
        if (i == 1 && (arg < 1 || arg > PHILO_MAX_COUNT))
        return 1;
        else if (i == 5 && (arg < 0 || arg > INT_MAX))
        return 1;
        else if (i == 5 && arg == 0)
        {
        return 2;
        }
        else if (i != 1 && i != 5 && (arg < 1 || arg > INT_MAX))
        return 1;
        i++;
    }
    return 0;
}#include "philosophers.h"

void *monitor(void *arg)
{
    t_general *data = (t_general *)arg;

    while (!data->dead)
    {
        int i = 0;
        while (i < data->number_of_philosophers || all_ate(data) != 0)
        {
            long long now = timestamp_ms();
            if ((now - data->philo[i].last_meal > data->time_to_die))
            {
                pthread_mutex_lock(&data->print);
                printf("%lld ms  Philosopher %d died\n",
                       (now - data->start_time) % 100000, data->philo[i].id);
                data->dead = 1;
                pthread_mutex_unlock(&data->print);
                return NULL;
            }
            i++;
        }
        usleep(1000);
    }
    return NULL;
}

void *ft_routine(void *arg)
{
    t_philosophers *p = (t_philosophers *)arg;

    if (p->id % 2 == 0)
        usleep(50);

    while (!p->data->dead)
    {
        take_forks(p);
        eat(p);
        release_forks(p);
        sleep_philo(p);
        think(p);
    }
    return NULL;
}

int all_ate(t_general *data)
{
    int ate;
    int i;
    int j;
    ate = 0;
    i = 0;
    j = 0;

    while(i < data->number_of_philosophers)
    {
        if (data->philo[i].times_eaten >= data->number_of_times_each_philosopher_must_eat)
        j++;
    i++;
    }   
    if (j == data->number_of_philosophers)
        ate++;
    return ate;
}#include "philosophers.h"

void error_handler(int err)
{   
    if (err == 1)
    printf("invalid/wrong number of arguments");
    else if (err == 2)
    printf("un filosofo no puede comer 0 veces");
}

void print_philos(t_general *data)
{
    int i = 0;
    while (i < data->number_of_philosophers)
    {
        printf("Philosopher [%d]: L=%d, R=%d\n",
               data->philo[i].id,
               data->philo[i].fork_l,
               data->philo[i].fork_r);
        i++;
    }
    printf("die=%lld, eat=%lld, sleep=%lld, times=%d\n",
           data->time_to_die, data->time_to_eat, data->time_to_sleep,
           data->number_of_times_each_philosopher_must_eat);
}

long long timestamp_ms(void)
{
    struct timeval tv;
    gettimeofday(&tv, NULL);
    return (tv.tv_sec * 1000 + tv.tv_usec / 1000);
}

void print_state(t_philosophers *philo, const char *msg)
{
    pthread_mutex_lock(&philo->data->print);
    if (!philo->data->dead)
    {
        long long now = timestamp_ms() - philo->data->start_time;
        printf("%lld %d %s\n", now, philo->id, msg);
    }
    pthread_mutex_unlock(&philo->data->print);
}


long long getime_now(void)
{
    long long start;
    long long new_start;
    struct timeval tv;

    start = timestamp_ms(); 
    new_start = gettimeofday(&tv, NULL)* 1000;
    return (new_start - start);
}

void smart_sleep(long long duration_ms, t_general *data)
{
    long long start = timestamp_ms();
    while (!data->dead)
    {
        if (timestamp_ms() - start >= duration_ms)
            break;
        usleep(100); // duermes en tramos pequeños
    }
}
